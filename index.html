Code Example — Data Pull

import pandas as pd

def fetch_monthly_revenue(csv_path: str) -> pd.DataFrame:
    df = pd.read_csv(csv_path, parse_dates=['date'])
    df = df.set_index('date').resample('M').sum()
    return df

# Usage
revenue = fetch_monthly_revenue('data/revenue.csv')
print(revenue.tail())
